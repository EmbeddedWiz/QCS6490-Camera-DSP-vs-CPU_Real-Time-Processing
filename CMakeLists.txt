cmake_minimum_required(VERSION 3.16)
project(QCS6490CameraDspVerdict LANGUAGES C CXX)

#
# The demo: one camera frame goes either to the CDSP or to the CPU, the
# result is presented through waylandsink, and the control panel reports
# frame time and the AGC state.
#
# The DSP half is a Hexagon shared object (dsp/libpipe_hvx_skel.so) built on
# a machine with the Hexagon SDK; it is shipped prebuilt here so the demo
# builds and runs on the board alone.  How it is made: docs/dsp-build.md.
#

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

add_compile_options(-Wall -Wextra)

find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET
    gstreamer-1.0
    gstreamer-app-1.0
    gstreamer-video-1.0
    gstreamer-allocators-1.0)

add_executable(pipeline
    src/main.cpp
    src/v4l2_capture.cpp
    src/route.cpp
    src/sensor.cpp
    src/dsp_engine.cpp
    src/cpu_engine.cpp
    src/sink.cpp
    src/panel.cpp
    idl/pipe_hvx_stub.c
)

target_include_directories(pipeline PRIVATE
    src
    idl
    /usr/include/fastrpc
)

target_link_libraries(pipeline PRIVATE
    PkgConfig::GST
    cdsprpc
)

# pipe_hvx_stub.c is generated by the Hexagon SDK; its type-punning and its
# size checks are the SDK's, not ours, so it is built without warnings.
set_source_files_properties(idl/pipe_hvx_stub.c PROPERTIES COMPILE_OPTIONS "-w")

# A synthetic frame for --parity / --replay, so the numbers can be reproduced
# without the camera.
add_executable(make_frame tools/make_frame.cpp)

# The binary looks for the Hexagon skeleton next to itself (build/dsp), so a
# plain `./build/pipeline` finds it without any environment variable.
add_custom_command(TARGET pipeline POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/dsp
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            ${CMAKE_CURRENT_SOURCE_DIR}/dsp/libpipe_hvx_skel.so
            ${CMAKE_BINARY_DIR}/dsp/libpipe_hvx_skel.so
)
